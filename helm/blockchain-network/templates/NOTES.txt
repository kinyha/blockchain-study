Useful commands:

  # Check pods status
  kubectl get pods -n {{ .Release.Namespace }} -l app.kubernetes.io/instance={{ .Release.Name }}

  # View logs
  kubectl logs -f deployment/{{ include "blockchain-network.fullname" . }}-miner -n {{ .Release.Namespace }}

  # Check blockchain height
  curl http://miner.localhost/api/blocks/height

  # Mine a block
  curl -X POST http://miner.localhost/api/mining/mine

{{- if .Values.nodes.orchestrator.enabled }}

    -H "Content-Type: application/json" \
    -d '{"nodeId":"miner-0","url":"http://{{ include "blockchain-network.fullname" . }}-miner:8080","role":"miner"}'

  # Get network status
  curl http://orchestrator.localhost/api/orchestrator/network
